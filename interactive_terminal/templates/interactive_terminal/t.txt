<script>

    // CREATE TERMINAL
    const term = new Terminal({

        cursorBlink: true,

        fontSize: 16,

        theme: {

            background: "#000000",

            foreground: "#ffffff"
        }
    });

    // OPEN TERMINAL
    term.open(document.getElementById("terminal"));
    term.writeln("=================================");
    term.writeln("      VIRTUAL LAB TERMINAL");
    term.writeln("=================================");
    term.writeln("");

    // PROMPT
    term.write("Akshay@gpu010:~$ ");

    // STORE COMMAND
    let currentCommand = "";

    // SEND COMMAND
    async function sendCommand(command) {

        try {

            let response = await fetch(
                "/terminal/run-command/",
                {

                    method: "POST",

                    headers: {

                        "Content-Type":
                            "application/x-www-form-urlencoded",

                        "X-CSRFToken":
                            "{{ csrf_token }}"
                    },

                    body:
                        "command=" +
                        encodeURIComponent(command)
                }
            );

            let data = await response.json();

            let output = data.output;

            // REMOVE DUPLICATE COMMAND
            output = output.replace(command, "");

            // PRINT OUTPUT
            term.write("\r\n" + output);

            // NEW PROMPT
            term.write("\r\nAkshay@gpu010:~$ ");
        }

        catch(error) {

            term.write(
                "\r\nConnection Error\r\n"
            );

            term.write(
                "\r\nAkshay@gpu010:~$ "
            );
        }
    }

    // TERMINAL INPUT
    term.onData(function(data) {

        // ENTER
        if (data === "\r") {

            sendCommand(currentCommand);

            currentCommand = "";
        }

        // BACKSPACE
        else if (data === "\u007F") {

            if (currentCommand.length > 0) {

                currentCommand =
                    currentCommand.slice(0, -1);

                term.write("\b \b");
            }
        }

        // NORMAL CHARACTERS
        else {

            currentCommand += data;

            term.write(data);
        }
    });

</script>
